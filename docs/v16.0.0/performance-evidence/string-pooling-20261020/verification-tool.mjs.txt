async function inventory(base){
 const files=[];
 async function visit(dir){
  for(const item of await readdir(dir,{withFileTypes:true})){
   const path=resolve(dir,item.name);
   if(item.isDirectory())await visit(path);
   else{ assert.ok(item.isFile()); const b=await readFile(path);
    files.push({path:relative(base,path).replaceAll('\\','/'),bytes:b.length,sha256:sha(b)}); }
  }
 }
 await visit(base);files.sort((a,b)=>a.path.localeCompare(b.path));
 return {files:files.length,bytes:files.reduce((n,f)=>n+f.bytes,0),sha256:sha(JSON.stringify(files))};
}
