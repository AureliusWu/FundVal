import { createHash } from 'node:crypto';
import { execFileSync } from 'node:child_process';

const hash = bytes => createHash('sha256').update(bytes).digest('hex');
const allowedDirs = new Set(['artifacts', 'qualification-candidates']);
async function inventory(base) {
  const rows = [];
  async function visit(path) {
    assert.ok(!path.includes('\\') && path.split('/').every(part => part && part !== '..' && part !== '.'));
    const absolute = resolve(base, path), info = await lstat(absolute);
    assert.equal(info.isSymbolicLink(), false);
    if (info.isDirectory()) {
      if (!path.includes('/')) assert.ok(allowedDirs.has(path), 'unexpected source folder');
      for (const name of (await readdir(absolute)).sort()) await visit(path + '/' + name);
    } else {
      assert.ok(info.isFile(), 'regular files only');
      if (!path.includes('/')) assert.match(path, /^[A-Za-z0-9._-]+\.(?:json|mjs|txt|md)$/);
      const bytes = await readFile(absolute);
      rows.push({ path, bytes: bytes.length, sha256: hash(bytes) });
    }
  }
  for (const name of (await readdir(base)).sort()) await visit(name);
  return rows.sort((a, b) => a.path.localeCompare(b.path));
}
