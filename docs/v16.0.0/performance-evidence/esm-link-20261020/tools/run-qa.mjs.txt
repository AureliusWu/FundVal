const hash = bytes => createHash('sha256').update(bytes).digest('hex');
async function inventory(base) {
  const rows = [];
  async function walk(path) {
    for (const item of await readdir(path, { withFileTypes: true })) {
      const target = resolve(path, item.name);
      if (item.isDirectory()) await walk(target);
      else if (item.isFile()) { const bytes = await readFile(target); rows.push({ path: relative(base, target).replaceAll('\\', '/'), bytes: bytes.length, sha256: hash(bytes) }); }
      else throw Error('Unexpected inventory object');
    }
  }
  await walk(base); rows.sort((a, b) => a.path.localeCompare(b.path));
  return { files: rows.length, bytes: rows.reduce((n, r) => n + r.bytes, 0), jsonInventorySha256: hash(JSON.stringify(rows)) };
}
async function sourceIdentity() {
  const paths = execFileSync('git', ['ls-files', '--cached', '--others', '--exclude-standard'], { cwd: root, encoding: 'utf8' }).trim().split(/\r?\n/)
    .filter(path => /^(?:js\/|scripts\/|test\/|e2e\/|\.github\/|package(?:-lock)?\.json$|playwright\.config\.mjs$)/.test(path)).sort();
  const rows = [];
  for (const path of paths) rows.push({ path, sha256: hash((await readFile(resolve(root, path), 'utf8')).replaceAll('\r\n', '\n')) });
  return { count: rows.length, lfSha256: hash(JSON.stringify(rows)), rows };
}
const record = { schema: 1, businessCommit: execFileSync('git', ['rev-parse', 'HEAD'], { cwd: root, encoding: 'utf8' }).trim(),

async function persist() { await writeFile(resolve(directory, 'qa-receipt.json'), JSON.stringify(record, null, 2) + '\n'); }
await persist();
async function run(id, args, expectedExit = 0) {
  const log = resolve(directory, id + '.log'); const fd = openSync(log, 'wx');
  const startedAt = new Date().toISOString(), start = performance.now();
  const child = spawn(process.execPath, args, { cwd: root, env: { ...process.env, SOURCE_DATE_EPOCH: epoch, FUNDVAL_E2E_PORT: '12437' },
    stdio: ['ignore', fd, fd], windowsHide: true });
  const result = await new Promise((accept, reject) => { child.on('error', reject); child.on('close', (code, signal) => accept({ code, signal })); });
  closeSync(fd); const bytes = await readFile(log);
  const row = { id, executable: process.execPath, args, startedAtHost: startedAt, exitCode: result.code, signal: result.signal,
    elapsedMs: performance.now() - start, expectedExit, log: { path: relative(directory, log), bytes: bytes.length, sha256: hash(bytes) } };
  record.commands.push(row); await persist(); process.stdout.write(JSON.stringify(row) + '\n');
  if (result.code !== expectedExit) throw Error(id + ' unexpected exit');
  return bytes.toString('utf8');
}
