function inventory(base, excluded = new Set(), part = '') {
  const rows = [];
  for (const entry of readdirSync(resolve(base, part), { withFileTypes: true })) {
    if (!part && excluded.has(entry.name)) continue;
    const path = part ? part + '/' + entry.name : entry.name;
    const physical = resolve(base, path);
    assert.equal(lstatSync(physical).isSymbolicLink(), false, 'No symlink: ' + physical);
    if (entry.isDirectory()) rows.push(...inventory(base, excluded, path));
    else {
      assert.equal(entry.isFile(), true, 'Regular files only: ' + physical);
      const bytes = readFileSync(physical);
      rows.push({ path, bytes: bytes.length, sha256: sha(bytes) });
    }
  }
  return rows.sort((a, b) => a.path.localeCompare(b.path));
}
function verifyRows(base, rows) {
  for (const row of rows) {
    const bytes = readFileSync(resolve(base, row.path));
    assert.equal(bytes.length, row.bytes, row.path + ' bytes');
    assert.equal(sha(bytes), row.sha256, row.path + ' SHA');
  }
}
function native(label, command, args, timeout = 180000) {
  const startedAt = new Date().toISOString();
  const env = { ...process.env }; delete env.NODE_OPTIONS; delete env.NODE_V8_COVERAGE;
  const result = spawnSync(command, args, { cwd: here, windowsHide: true, encoding: null, maxBuffer: 30_000_000, timeout, env });
  const stdout = result.stdout ?? Buffer.alloc(0), stderr = result.stderr ?? Buffer.alloc(0);
  save(resolve(here, label + '.stdout'), stdout); save(resolve(here, label + '.stderr'), stderr);
  const receipt = { label, command, args, cwd: here, startedAt, finishedAt: new Date().toISOString(),
    nativeExit: result.status, signal: result.signal, error: result.error?.message ?? null,
    stdout: { bytes: stdout.length, sha256: sha(stdout) }, stderr: { bytes: stderr.length, sha256: sha(stderr) } };
  saveJSON(resolve(here, label + '.receipt.json'), receipt);
  assert.equal(receipt.nativeExit, 0, label + ' native exit');
  assert.equal(receipt.error, null); assert.equal(receipt.signal, null);
  return { receipt, stdout, stderr };
}
