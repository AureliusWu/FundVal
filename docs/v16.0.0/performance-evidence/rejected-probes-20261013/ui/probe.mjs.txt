function walk(node, visit) {
  if (!node || typeof node !== 'object') return;
  visit(node);
  for (const value of Object.values(node)) {
    if (Array.isArray(value)) value.forEach(child => walk(child, visit));
    else if (value && typeof value === 'object') walk(value, visit);
  }
}
function chain(node) {
  if (node.type !== 'BinaryExpression' || node.operator !== '+') return [node];
  return [...chain(node.left), node.right];
}

  walk(owner, node => {
    if (node.type !== 'AssignmentExpression' || node.operator !== '+=' || node.left.type !== 'Identifier' || node.left.name !== 'html') return;

  });
}

const templateText = value => value.replaceAll('\\', '\\\\').replaceAll('`', '\\`').replaceAll('${', '\\${');
function after(replacement, strict) {
  return '`' + replacement.pieces.map(piece => piece.literal ? templateText(piece.value)
    : '${' + (strict ? "'' + (" + piece.source + ')' : piece.source) + '}').join('') + '`';
}
function variant(strict) {
  let source = app;
  for (const replacement of [...replacements].sort((a,b) => b.start - a.start)) {
    source = source.slice(0, replacement.start) + after(replacement, strict) + source.slice(replacement.end);
  }
  const program = parse('app.js', source);
  assert.equal(program.body.length, appProgram.body.length);
  return source;
}

function staticClosure(byFile, roots) {
  const seen = new Set(), queue = [...roots];
  while (queue.length) {
    const filename = queue.shift();
    if (seen.has(filename)) continue;
    const chunk = byFile.get(filename);
    assert.ok(chunk, `Generated static dependency ${filename} exists`);
    seen.add(filename);
    queue.push(...chunk.imports);
  }
  return seen;
}
async function measure(name) {
  const buildConfig = config();
  let seen = 0;
  if (name !== 'baseline') buildConfig.plugins = [{ name: `fundval-memory-${name}`, enforce: 'pre',
    transform(source, id) {
      if (id.replaceAll('\\', '/') !== appPath.replaceAll('\\', '/')) return null;
      assert.equal(source.replaceAll('\r\n', '\n'), app);
      seen++;
      return { code: variants[name], map: null };
    } }];
  const result = await viteBuild(buildConfig);
  if (name !== 'baseline') assert.equal(seen, 1, 'Exactly one app module transformed');
  const outputs = Array.isArray(result) ? result.flatMap(item => item.output || []) : result.output || [];
  const chunks = outputs.filter(output => output.type === 'chunk');
  const entry = chunks.find(chunk => chunk.isEntry);
  assert.equal(entry.fileName, 'js/app-shell.js');
  const byFile = new Map(chunks.map(chunk => [chunk.fileName, chunk]));
  const cold = staticClosure(byFile, [entry.fileName, ...entry.dynamicImports]);
  const coldModules = [...cold].flatMap(filename => Object.keys(byFile.get(filename).modules));
  assert.ok(!coldModules.some(id => /(?:ocr|paddle|onnx|tesseract|\/storage\/(?:cloud-|gist-)|\/runtime\/(?:quote-diagnostics|diagnostics-ui))/.test(id)));
  const inventory = chunks.map(chunk => ({ bytes: Buffer.byteLength(chunk.code), gzip: gzipSync(chunk.code).length,
    role: cold.has(chunk.fileName) ? 'cold' : 'lazy', modules: Object.keys(chunk.modules).map(id => relative(root, id).replaceAll('\\','/')).sort() }));
  const graph = chunks.map(chunk => ({ modules: Object.keys(chunk.modules).map(id => relative(root,id).replaceAll('\\','/')).sort(),
    role: cold.has(chunk.fileName) ? 'cold' : 'lazy',
    imports: chunk.imports.map(id => Object.keys(byFile.get(id).modules).map(moduleId => relative(root,moduleId).replaceAll('\\','/')).sort()),
    dynamic: chunk.dynamicImports.map(id => Object.keys(byFile.get(id).modules).map(moduleId => relative(root,moduleId).replaceAll('\\','/')).sort())
  })).sort((a,b) => JSON.stringify(a.modules).localeCompare(JSON.stringify(b.modules)));
  return { chunks: chunks.length, raw: inventory.reduce((a,b) => a+b.bytes,0),
    all: inventory.reduce((a,b) => a+b.gzip,0), cold: inventory.filter(item=>item.role==='cold').reduce((a,b)=>a+b.gzip,0), graphSha: sha(JSON.stringify(graph)) };
}
