function replaceOne(source, search, replacement) {
  const start = source.indexOf(search);
  assert.ok(start >= 0, `Missing replacement ${search.slice(0, 100)}`);
  assert.equal(source.indexOf(search, start + search.length), -1, `Nonunique replacement ${search.slice(0, 100)}`);
  return source.slice(0, start) + replacement + source.slice(start + search.length);
}

function migrationSuccess(source) {
  const helper = `function successfulHoldingsParse(document, sourceSchema, migrated) {
  return Object.freeze({ ok: true, sourceSchema, readonly: false, migrated, empty: document.holdings.length === 0, document });
}

`;
  source = replaceOne(source, 'export function parseAndMigrateHoldings(value, options = {}) {', helper + 'export function parseAndMigrateHoldings(value, options = {}) {');
  source = replaceOne(source,
    'return Object.freeze({ ok: true, sourceSchema: 1, readonly: false, migrated: true, empty: document.holdings.length === 0, document });',
    'return successfulHoldingsParse(document, 1, true);');
  source = replaceOne(source,
    'return Object.freeze({ ok: true, sourceSchema, readonly: false, migrated: false, empty: document.holdings.length === 0, document });',
    'return successfulHoldingsParse(document, sourceSchema, false);');
  return replaceOne(source,
    'return Object.freeze({ ok: true, sourceSchema, readonly: false, migrated: true, empty: document.holdings.length === 0, document });',
    'return successfulHoldingsParse(document, sourceSchema, true);');
}

function schemaPrivateScalar(source) {
  source = replaceOne(source,
    "function text(value, field, { fallback = '', required = false, maximum = MAX_TEXT, nullable = false } = {}) {",
    "function text(value, field, fallback = '', required = false, maximum = MAX_TEXT, nullable = false) {");
  const replacements = [
    ["text(value, field, { required: true, maximum: 80 })", "text(value, field, '', true, 80)"],
    ["text(value.fundCode, 'fundCode', { required: true, maximum: 6 })", "text(value.fundCode, 'fundCode', '', true, 6)"],
    ["text(value.id, 'id', { required: true, maximum: 120 })", "text(value.id, 'id', '', true, 120)"],
    ["text(value.fundName, 'fundName', { fallback: fundCode, maximum: 120 })", "text(value.fundName, 'fundName', fundCode, false, 120)"],
    ["text(value.note, 'note', { maximum: 500, nullable: true })", "text(value.note, 'note', '', false, 500, true)"],
    ["function nonNegativeNumber(value, field, { nullable = false } = {}) {", "function nonNegativeNumber(value, field, nullable = false) {"],
    ["nonNegativeNumber(value.costNav, 'costNav', { nullable: true })", "nonNegativeNumber(value.costNav, 'costNav', true)"],
  ];
  for (const [search, replacement] of replacements) source = replaceOne(source, search, replacement);
  const deviceSearch = "text(value.deviceId, 'deviceId', { required: true, maximum: 120 })";
  assert.equal(source.split(deviceSearch).length - 1, 2);
  return source.replaceAll(deviceSearch, "text(value.deviceId, 'deviceId', '', true, 120)");
}

function collect(chunks, roots) {
  const byName = new Map(chunks.map(chunk => [chunk.fileName, chunk]));
  const included = new Set();
  const queue = [...roots];
  while (queue.length) {
    const file = queue.shift();
    if (included.has(file)) continue;
    included.add(file);
    assert.ok(byName.has(file));
    queue.push(...byName.get(file).imports);
  }
  return included;
}
