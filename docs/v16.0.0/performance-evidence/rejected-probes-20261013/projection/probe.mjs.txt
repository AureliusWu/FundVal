  for (const chunk of chunks) {
    assert.ok(/^js\/(?:app-shell|chunks\/[A-Za-z0-9._-]+)\.js$/.test(chunk.fileName));
    assert.ok(!tools.OCR_BUNDLE_REFERENCE.test(chunk.code));

    coldGzip: chunks.filter(c => cold.has(c.fileName)).reduce((sum, c) => sum + gzipSync(c.code).length, 0),
