const sha = bytes => createHash('sha256').update(bytes).digest('hex');
const git = (...args) => execFileSync('git', args, { cwd: root, encoding: 'utf8' }).trim();
function sourceInventory() {
  const paths = [...new Set(git('ls-files', '--cached', '--others', '--exclude-standard').split('\n'))]
    .filter(p => /^(?:js|scripts|test|e2e|data|css)\//.test(p) || /^(?:index.html|sw.js|manifest.json|package.json|package-lock.json|playwright.config.mjs)$/.test(p)).sort();
  const files = paths.map(path => { const b = readFileSync(join(root, path)); return { path, bytes: b.length, sha256: sha(b) }; });
  return { files: files.length, bytes: files.reduce((n, f) => n + f.bytes, 0), sha256: sha(JSON.stringify(files)) };
}
function siteInventory() {
  const paths = [];
  function walk(dir) {
    for (const entry of readdirSync(dir, { withFileTypes: true })) {
      const file = join(dir, entry.name);
      if (entry.isDirectory()) walk(file);
      else if (entry.isFile()) paths.push(relative(join(root, 'site'), file).replaceAll('\\', '/'));
      else throw Error('Unexpected nonregular site entry');
    }
  }
  walk(join(root, 'site'));
  const files = paths.sort((a,b) => a.localeCompare(b)).map(path => {
    const b = readFileSync(join(root, 'site', path));
    return { path, bytes: b.length, sha256: sha(b) };
  });
  return { files: files.length, bytes: files.reduce((n,f) => n + f.bytes, 0), sha256: sha(JSON.stringify(files)) };
}

function run(name, command, expected = 0) {
  const start = performance.now();
  const result = spawnSync('powershell.exe', ['-NoProfile','-Command', command], { cwd: root, env, encoding: 'utf8', maxBuffer: 64*1024*1024 });
  const log = String(result.stdout || '') + String(result.stderr || '');
  writeFileSync(join(output, name + '.log'), log, { flag: 'wx' });
  const record = { name, command, exit: result.status, expected, durationMs: performance.now() - start, log: name + '.log' };
  report.commands.push(record);
  process.stdout.write(JSON.stringify(record) + '\n');
  if (result.error || result.status !== expected) throw Error(name + ' unexpected result: ' + log.slice(-3000));
  return log;
}
