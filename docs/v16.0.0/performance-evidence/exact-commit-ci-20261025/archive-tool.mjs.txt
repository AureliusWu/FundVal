function hashFile(path) {
  const h = createHash('sha256'), fd = openSync(path, 'r'), buf = Buffer.allocUnsafe(1024 * 1024);
  let bytes = 0; try { for (;;) { const n = readSync(fd, buf, 0, buf.length, null); if (!n) break; h.update(buf.subarray(0, n)); bytes += n; } } finally { closeSync(fd); }
  return { bytes, sha256: h.digest('hex') };
}
function topInventory(base) {
  assert.equal(lstatSync(base).isDirectory(), true); assert.equal(lstatSync(base).isSymbolicLink(), false);
  return readdirSync(base, { withFileTypes: true }).map(ent => {
    const path = resolve(base, ent.name), stat = lstatSync(path);
    if (stat.isSymbolicLink()) return { path: ent.name, kind: 'link-not-followed', bytes: stat.size };
    if (stat.isDirectory()) return { path: ent.name, kind: 'directory-not-recursed' };
    assert.equal(stat.isFile(), true, 'Only regular files, directories and links permitted in source metadata');
    return { path: ent.name, kind: 'file', ...hashFile(path) };
  }).sort((a, b) => a.path.localeCompare(b.path));
}
function inventory(base, part = '') {
  const rows = [];
  for (const entry of readdirSync(resolve(base, part), { withFileTypes: true })) {
    const path = part ? part + '/' + entry.name : entry.name, physical = resolve(base, path), stat = lstatSync(physical);
    assert.equal(stat.isSymbolicLink(), false, 'No stage/extract links');
    if (stat.isDirectory()) rows.push(...inventory(base, path));
    else { assert.equal(stat.isFile(), true); rows.push({ path, ...hashFile(physical) }); }
  }
  return rows.sort((a, b) => a.path.localeCompare(b.path));
}
function native(label, command, args, timeout = 180000) {
  const startedAt = new Date().toISOString();
  const env = { ...process.env }; delete env.NODE_OPTIONS; delete env.NODE_V8_COVERAGE;
  const r = spawnSync(command, args, { cwd: here, windowsHide: true, encoding: null, maxBuffer: 40 * 1024 * 1024, timeout, env });
  const stdout = r.stdout ?? Buffer.alloc(0), stderr = r.stderr ?? Buffer.alloc(0);
  save(resolve(here, label + '.stdout'), stdout); save(resolve(here, label + '.stderr'), stderr);
  const receipt = { command, args, cwd: here, startedAt, finishedAt: new Date().toISOString(), nativeExit: r.status, signal: r.signal, error: r.error?.message ?? null,
    stdout: { bytes: stdout.length, sha256: sha(stdout) }, stderr: { bytes: stderr.length, sha256: sha(stderr) } };
  saveJSON(resolve(here, label + '.receipt.json'), receipt);
  assert.equal(r.status, 0, label + ' native exit'); assert.equal(r.signal, null); assert.equal(r.error, undefined);
  return { receipt, stdout, stderr };
}
function gitGuard(label) {
  return Object.fromEntries([
    ['head', ['rev-parse', 'HEAD']], ['branch', ['branch', '--show-current']],
    ['index', ['diff', '--cached', '--binary']],
    ['tracked-diff', ['diff', '--binary', 'HEAD', '--', '.', ':!docs/v16.0.0/performance-evidence/exact-commit-ci-20261025']],
  ].map(([tag, args]) => {
    const r = native(label + '-' + tag, 'git', ['-C', root, ...args]);
    return [tag, { bytes: r.stdout.length, sha256: sha(r.stdout), ...(tag === 'head' || tag === 'branch' ? { value: r.stdout.toString('utf8').trim() } : {}) }];
  }));
}

function scan(bytes, allowedFixture = false) {
  // latin1 indexes equal original byte offsets, preserving non-UTF8 native diagnostics.
  const text = bytes.toString('latin1'), findings = [], reviewed = [];
  const patterns = [
    ['high-confidence-secret', /(?:gh[pousr]_[A-Za-z0-9]{30,}|github_pat_[A-Za-z0-9_]{40,}|-----BEGIN (?:RSA |EC |OPENSSH )?PRIVATE KEY-----)/g],
    ['non-noreply-email', /[A-Za-z0-9.!#$%&'*+/=?^_`{|}~-]+@[A-Za-z0-9](?:[A-Za-z0-9.-]*[A-Za-z0-9])?\.[A-Za-z]{2,63}/g],
    ['signed-or-credential-url', /https?:\/\/[^\s<>"'`]+/g],
  ];
  for (const [kind, pattern] of patterns) for (const m of text.matchAll(pattern)) {
    if (kind === 'non-noreply-email' && /@users\.noreply\.github\.com$/i.test(m[0])) continue;
    if (kind === 'signed-or-credential-url' && !/[?&](?:X-Amz-[^=\s]+|sig(?:nature)?|token|access_token|jwt|se|sv|sp)=/i.test(m[0]) && !/^https?:\/\/[^/]+@/.test(m[0])) continue;
    const hit = { kind, start: m.index, end: m.index + m[0].length, bytes: m[0].length, sha256: sha(Buffer.from(m[0], 'latin1')) };
    if (allowedFixture && kind === 'high-confidence-secret' && approvedFixtureMatches.has(hit.sha256)) reviewed.push(hit);
    else findings.push(hit);
  }
  if (emailValue && !gitEmailIsPublicNoreply) {
    const needle = Buffer.from(emailValue, 'utf8'); let at = 0;
    while ((at = bytes.indexOf(needle, at)) !== -1) { findings.push({ kind: 'actual-private-git-email', start: at, end: at + needle.length, bytes: needle.length, sha256: sha(needle) }); at += needle.length; }
  }
  return { findings, reviewed };
}

function redact(bytes, hits) {
  const ordered = hits.slice().sort((a, b) => a.start - b.start || b.end - a.end), ranges = [];
  for (const hit of ordered) { const prev = ranges.at(-1); if (prev && hit.start <= prev.end) prev.end = Math.max(prev.end, hit.end); else ranges.push({ start: hit.start, end: hit.end }); }
  let at = 0; const out = [Buffer.from('DERIVED REDACTED COPY — NOT ORIGINAL NATIVE BYTES. Original byte count/SHA and findings are in archive-plan.json.\n', 'utf8')];
  for (const r of ranges) { out.push(bytes.subarray(at, r.start), Buffer.from('[REDACTED-PRIVATE-OR-SECRET-SHAPED-VALUE]')); at = r.end; }
  out.push(bytes.subarray(at)); return Buffer.concat(out);
}

      omissions.push({ source: archivePath, ...row, reason: row.path === 'exact-source.tar' ?     }
    assert.equal(/\.(?:jpg|jpeg|png|webp|gif|sqlite|db)$/i.test(row.path), false, 'Unexpected screenshot or user-data-shaped payload');
    const bytes = readFileSync(resolve(input.originalPath, row.path)); assert.equal(bytes.length, row.bytes); assert.equal(sha(bytes), row.sha256);
    const report = input.prefix === 'exact-v1-sourceprobe' && row.path === 'probe-git-blobs.stdout' ? probeScreen(input.originalPath, bytes) : scan(bytes);

      privacy: 'All non-noreply email/signed URL/high-confidence token hits excluded as raw public bytes; inventory original SHA plus explicitly labeled derived redacted bytes only. Current Git email stays memory-only.',

}
