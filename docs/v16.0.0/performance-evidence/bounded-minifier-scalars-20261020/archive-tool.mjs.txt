async function visit(base, path, group) {
  if (path.includes('\\') || path.split('/').some(part => !part || part === '..' || part === '.')) throw Error('Unsafe archive path');
  const file = resolve(base, path), metadata = await lstat(file);
  if (metadata.isSymbolicLink()) throw Error('Refusing archive symlink');
  if (metadata.isDirectory()) {
    for (const name of (await readdir(file)).sort()) await visit(base, path + '/' + name, group);
  } else if (metadata.isFile()) {
    if (seen.has(path)) throw Error('Archive input path collision'); seen.add(path);
    const bytes = await readFile(file); inventory.push({ path, group, bytes: bytes.length, sha256: hash(bytes) });
  } else throw Error('Unexpected archive input');
}
